create function fn_nome(valor numeric)
returns varchar
language plpgsql
as $$
	begin
		if(valor > 18) then
			return 'maior de idade';
		elsif(valor = 18) then
			return 'exatamente 18 anos';
		else
			return 'menor de idade';
		end if;
	end
$$;

select *,
	fn_nome(id_atividade) from tb_atividade;

	extract (year from age(dt_nascimento))

create function fn_verificarIdade (idade date)
returns varchar
language plpgsql
as $$ 
declare idade integer;
	begin
		if (extract (year from age(idade)) > 18) then
		return 'Maior de Idade';

		elsif (extract (year from age(idade)) = 18) then
		return 'Exatamente 18 anos de Idade';
		else return 'Menor de Idade';
		end if;	
	end
$$;

select *,
fn_verificarIdade(dt_nascimento)
from
tb_atividade;

create function fc_calculadora (valorUm numeric, valorDois numeric, operador varchar(1))
returns numeric
language plpgsql
as $$
	begin
		if (operador = '-') then 
			return valorUm - valorDois;
		elsif (operador = '+') then
			return valorUm + valorDois;
		elsif (operador = '*') then
			return valorUm * valorDois;
		else 
			return valorUm / valorDois;
		end if;
	end;
$$;

select fc_calculadora (2, 4, '*')
from tb_atividade;

create function fc_baratocaro (preco numeric)
returns varchar
language plpgsql
as $$
begin
	if (preco < 50) then
		return 'Barato';
	elsif (preco < 200) then
		return 'Médio';
		else return 'Caro';
	end if;
end;
$$;

create function fc_notas (nota numeric)
returns numeric
language plpgsql
as $$
begin
	if (valor < 0 or valor > 10) then
		return 'nota inválida. insira um número menor que 10 ou maior que 0.';
	elsif (valor >= 7) then
		return 'aprovado';
	elsif (valor < 7 or valor >= 5) then 
		return 'recuperação';
	else return 'reprovado';
end if;
end;
$$;

create or replace view vw_simples as(  
select 
	prod.id_produto as codigo_produto,
	prod.tx_descricao as descricao,
	pedi.vl_total as total,
	max(dt_pedido) as maiordataped,
	max(id_usuario) as maiorusuario
from tb_pedidos pedi
inner join tb_usuarios use
	on use.id_usuario = pedi.cd_usuario
inner join tb_itens_pedido item
	on pedi.id_pedido = item.cd_pedido
inner join tb_produtos prod
	on prod.id_produto = item.cd_produto
where
	exists (select 1 from tb_enderecos ende where tx_uf = upper('sc') and  ende.cd_usuario = use.id_usuario)
	and pedi.tx_status = upper ('pago')
	and pedi.vl_total > 0
group by prod.id_produto, prod.tx_descricao, pedi.vl_total
limit 50
);

select * from vw_simples;

CREATE OR REPLACE FUNCTION fn_estado (uf varchar)
RETURNS varchar
LANGUAGE plpgsql
AS $$
BEGIN
	return
    CASE upper (uf)
        WHEN 'SC' THEN 'SANTA CATARINA'
        WHEN 'SP' THEN 'SÃO PAULO'
		WHEN 'RS' THEN 'RIO GRANDE DO SUL'
        when 'PR' then 'PARANÁ'
		when 'RJ' then 'RIO DE JANEIRO'
    end case;
end;
$$;

select tx_uf, fn_estado(tx_uf) from tb_enderecos;
